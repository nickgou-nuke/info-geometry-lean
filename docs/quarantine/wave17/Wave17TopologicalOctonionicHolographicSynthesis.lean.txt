/-!
## Section 1: Commuting Bulk-to-Boundary Representation Square for $\operatorname{Der}(\mathbb{S}_s)$
-/

/--
Chiral boundary projection on `SplitSedenion ℝ`, projecting a 16D split-sedenion pair `(X.1, X.2)`
onto its positive chiral split-octonion boundary sector `(X.1, zornZero)`.
-/
def sedenionChiralBoundaryProj (X : SplitSedenion ℝ) : SplitSedenion ℝ :=
  (X.1, zornZero)

/--
Chiral boundary projection transported across `spinor16SplitSedenionEquiv` to the 16-component
spinor carrier `Fin 16 → ℝ`.
-/
def spinor16ChiralBoundaryProj (v : Fin 16 → ℝ) : Fin 16 → ℝ :=
  spinor16SplitSedenionEquiv.symm (sedenionChiralBoundaryProj (spinor16SplitSedenionEquiv v))

/--
`sedenionChiralBoundaryProj` is idempotent on `SplitSedenion ℝ`.
-/
theorem sedenionChiralBoundaryProj_idem (X : SplitSedenion ℝ) :
    sedenionChiralBoundaryProj (sedenionChiralBoundaryProj X) = sedenionChiralBoundaryProj X := by
  have _ := X
  rfl

/--
**Split-Sedenion Bulk-to-Boundary Intertwining Theorem**:
Every 14-parameter $G_2$ derivation `sedenionG2Derivation p` commutes strictly with the chiral
boundary projection `sedenionChiralBoundaryProj`.
-/
theorem sedenionChiralBoundaryProj_intertwines_g2Derivation
    (p : Params) (X : SplitSedenion ℝ) :
    sedenionChiralBoundaryProj (sedenionG2Derivation p X) =
      sedenionG2Derivation p (sedenionChiralBoundaryProj X) := by
  apply Prod.ext
  · rfl
  · dsimp [sedenionChiralBoundaryProj, sedenionG2Derivation, cdLift]
    exact (zornG2Derivation_zornZero p).symm

/--
**Spinor-16 Bulk-to-Boundary Commuting Representation Square**:
On the 16-component spinor space `Fin 16 → ℝ`, the transported $G_2$ derivation representation
`spinor16G2Derivation p` commutes with the chiral boundary projection `spinor16ChiralBoundaryProj`.
-/
theorem spinor16ChiralBoundaryProj_intertwines_g2Derivation
    (p : Params) (v : Fin 16 → ℝ) :
    spinor16ChiralBoundaryProj (spinor16G2Derivation p v) =
      spinor16G2Derivation p (spinor16ChiralBoundaryProj v) := by
  unfold spinor16ChiralBoundaryProj spinor16G2Derivation
  rw [spinor16SplitSedenionEquiv.apply_symm_apply,
      spinor16SplitSedenionEquiv.apply_symm_apply,
      sedenionChiralBoundaryProj_intertwines_g2Derivation p (spinor16SplitSedenionEquiv v)]

/--
Every projected spinor `spinor16ChiralBoundaryProj v` lies in the positive chiral semi-spinor
boundary subspace `SemiSpinorPlus`.
-/
theorem spinor16ChiralBoundaryProj_mem_semiSpinorPlus (v : Fin 16 → ℝ) :
    spinor16ChiralBoundaryProj v ∈ SemiSpinorPlus := by
  rw [mem_semiSpinorPlus_iff_snd_zero]
  change (spinor16SplitSedenionEquiv (spinor16ChiralBoundaryProj v)).2 = zornZero
  unfold spinor16ChiralBoundaryProj
  rw [spinor16SplitSedenionEquiv.apply_symm_apply]
  rfl

## Section 2: Akivis–Malcev Nonassociative Jacobiator Anomaly vs. Derivation Lie Closure
-/

/--
**Akivis–Malcev Associator Anomaly vs. Derivation Lie Bracket Closure**:
While the internal commutator of three quark elements in `SplitOctonion R` has a non-vanishing
Jacobiator equal to $6$ times their associator (`-6 det(u₁, u₂, u₃) • K₀`), the commutator
`sedenionDerivationCommutator` of any two 14-parameter $G_2$ $\star$-derivations on `SplitSedenion ℝ`
is again a genuine $\star$-derivation.
-/
theorem akivis_malcev_vs_derivation_lie_closure
    {R : Type*} [CommRing R]
    (p₁ p₂ : Params)
    (u₁ u₂ u₃ : MC.ColorVec3 R) :
    IsSedenionStarDerivation
      (sedenionDerivationCommutator (sedenionG2Derivation p₁) (sedenionG2Derivation p₂)) ∧
    MC.SplitOctonion.jacobiator
      (MC.SplitOctonion.quark u₁) (MC.SplitOctonion.quark u₂) (MC.SplitOctonion.quark u₃) =
      MC.SplitOctonion.scale 6
        (MC.SplitOctonion.associator
          (MC.SplitOctonion.quark u₁) (MC.SplitOctonion.quark u₂) (MC.SplitOctonion.quark u₃)) ∧
    MC.SplitOctonion.jacobiator
      (MC.SplitOctonion.quark u₁) (MC.SplitOctonion.quark u₂) (MC.SplitOctonion.quark u₃) =
      MC.SplitOctonion.scale (- 6 * MC.ColorVec3.det3 u₁ u₂ u₃)
        MC.SplitOctonion.paraSymplecticK0 :=
  ⟨isSedenionStarDerivation_commutator
     (sedenionG2Derivation p₁) (sedenionG2Derivation p₂)
     (sedenionG2Derivation_isStarDerivation p₁)
     (sedenionG2Derivation_isStarDerivation p₂),
   (MC.SplitOctonion.akivis_malcev_maurer_cartan_anomaly_theorem u₁ u₂ u₃).1,
   (MC.SplitOctonion.akivis_malcev_maurer_cartan_anomaly_theorem u₁ u₂ u₃).2⟩

## Section 4: Master Wave 17 Synthesis Theorem

/--
**Master Wave 17 Topological-Octonionic-Holographic Synthesis Theorem**:
Unifies in one theorem:

   (`sedenionChiralBoundaryProj_intertwines_g2Derivation` and

    (c1 c2 c3 : R) (X_bulk : Holo.SplitOctonion R)
    (J_source O_condensate : Holo.ColorVec3 R) :
    (sedenionChiralBoundaryProj (sedenionG2Derivation p₁ X_sed) =
       sedenionG2Derivation p₁ (sedenionChiralBoundaryProj X_sed) ∧
     spinor16ChiralBoundaryProj (spinor16G2Derivation p₁ v_spin) =
       spinor16G2Derivation p₁ (spinor16ChiralBoundaryProj v_spin)) ∧

      (sedenionDerivationCommutator (sedenionG2Derivation p₁) (sedenionG2Derivation p₂)) ∧

      ⟨DW.SplitOctonion.neg (DW.gaugeVarPhi F_brst).A,
       DW.SplitOctonion.zero, DW.SplitOctonion.zero, DW.SplitOctonion.zero⟩ ∧
    (ψ₁ * barψ₁ - ψ₂ * barψ₂)^2 + (2 * ψ₁ * barψ₂) * (2 * ψ₂ * barψ₁) =
      (ψ₁ * barψ₁ + ψ₂ * barψ₂)^2 ∧
    MC.SplitOctonion.jacobiator
      (MC.SplitOctonion.quark u₁) (MC.SplitOctonion.quark u₂) (MC.SplitOctonion.quark u₃) =
      MC.SplitOctonion.scale (- 6 * MC.ColorVec3.det3 u₁ u₂ u₃)
        MC.SplitOctonion.paraSymplecticK0 ∧

       (WJ.Mat3.mul (WJ.tlGenE1 (WJ.unknotVacuumLoop CS)) (WJ.tlGenE2 (WJ.unknotVacuumLoop CS)))

       Holo.bulkColorAction c1 c2 c3 (Holo.bulkToBoundaryChiralMap X_bulk) ∧
     Holo.boundarySourceResponsePairing J_source O_condensate =
       Holo.SplitOctonion.scale (Holo.ColorVec3.dot J_source O_condensate) Holo.SplitOctonion.one) :=
  ⟨⟨sedenionChiralBoundaryProj_intertwines_g2Derivation p₁ X_sed,
    spinor16ChiralBoundaryProj_intertwines_g2Derivation p₁ v_spin⟩,
   isSedenionStarDerivation_commutator
     (sedenionG2Derivation p₁) (sedenionG2Derivation p₂)
     (sedenionG2Derivation_isStarDerivation p₁)
     (sedenionG2Derivation_isStarDerivation p₂),
   DW.donaldson_witten_brst_sq_eq_neg_gauge_var F_brst,
   DW.swQuadraticMap_norm_sq_factorization ψ₁ ψ₂ barψ₁ barψ₂,
   (MC.SplitOctonion.akivis_malcev_maurer_cartan_anomaly_theorem u₁ u₂ u₃).2,
   ⟨WJ.tlGen_jones_cubic_121 (WJ.unknotVacuumLoop CS),
    WJ.kauffmanBraid_yang_baxter_121_eq_212 CS h_inv⟩,
   ⟨Holo.bulkToBoundary_intertwines_colorAction c1 c2 c3 X_bulk,
    Holo.boundarySourceResponsePairing_eq_scalar_vacuum J_source O_condensate⟩⟩
