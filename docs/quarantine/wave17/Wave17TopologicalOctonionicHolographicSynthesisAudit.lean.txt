/-- Concrete integer audit of the Donaldson-Witten equivariant BRST square relation `Q² = -δ_ϕ`. -/
theorem audit_wave17_donaldson_witten_brst_sq_int :
    let F : DW.BRSTFieldMultiplet (DW.SplitOctonion ℤ) :=
      ⟨DW.canonicalSplitOctonionInt, DW.SplitOctonion.one, DW.canonicalSplitOctonionInt, DW.SplitOctonion.zero⟩
    DW.brstQ (DW.brstQ F) =
      ⟨DW.SplitOctonion.neg (DW.gaugeVarPhi F).A,
       DW.SplitOctonion.zero, DW.SplitOctonion.zero, DW.SplitOctonion.zero⟩ := by
  decide

/-- Concrete integer audit of the Temperley-Lieb Jones cubic and Kauffman Yang-Baxter braid relations. -/
theorem audit_wave17_jones_yang_baxter_int :
    let CS := WJ.canonicalChernSimonsPhaseInt
    WJ.Mat3.mul
      (WJ.Mat3.mul (WJ.tlGenE1 (WJ.unknotVacuumLoop CS)) (WJ.tlGenE2 (WJ.unknotVacuumLoop CS)))
      (WJ.tlGenE1 (WJ.unknotVacuumLoop CS)) =
      WJ.tlGenE1 (WJ.unknotVacuumLoop CS) ∧
    WJ.Mat3.mul (WJ.Mat3.mul (WJ.kauffmanBraidR1 CS) (WJ.kauffmanBraidR2 CS)) (WJ.kauffmanBraidR1 CS) =
      WJ.Mat3.mul (WJ.Mat3.mul (WJ.kauffmanBraidR2 CS) (WJ.kauffmanBraidR1 CS)) (WJ.kauffmanBraidR2 CS) := by
  decide
