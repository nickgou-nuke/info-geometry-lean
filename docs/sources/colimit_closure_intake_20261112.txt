def comm {A : Type*} [Ring A] (X Y : A) : A := X * Y - Y * X

lemma invOf_two_comm {A : Type*} [Ring A] [Invertible (2 : A)] (x : A) :
    ⅟(2 : A) * x = x * ⅟(2 : A) := by
  have h1 : (2 : A) * ⅟(2 : A) = 1 := mul_invOf_self (2 : A)
  have h2 : ⅟(2 : A) * (2 : A) = 1 := invOf_mul_self (2 : A)

  calc ⅟(2 : A) * x

    _ = 1 * x * ⅟(2 : A) := by rw [h2]
    _ = x * ⅟(2 : A) := by noncomm_ring

lemma star_invOf_two {A : Type*} [Ring A] [StarRing A] [Invertible (2 : A)] :
    star (⅟(2 : A)) = ⅟(2 : A) := by

/-- Single-channel GKSL dissipator D_L(X) = L* X L - ½(L* L X + X L* L) in any *-ring with ⅟2. -/
def gkslDissipator {A : Type*} [Ring A] [StarRing A] [Invertible (2 : A)] (L X : A) : A :=
  star L * X * L - ⅟(2 : A) * (star L * L * X + X * star L * L)

lemma star_gkslDissipator {A : Type*} [Ring A] [StarRing A] [Invertible (2 : A)] (L X : A) :
    star (gkslDissipator L X) = gkslDissipator L (star X) := by
  dsimp [gkslDissipator]
  simp only [star_sub, star_add, star_mul, star_star, star_invOf_two]
  rw [← invOf_two_comm]
  noncomm_ring

/-- Reversible modular/Hamiltonian derivation [H, ·] for skew-adjoint H = J K_eff (H* = -H)
    has identically zero Kadison–Schwarz dissipation defect. -/
theorem reversible_derivation_zero_defect {A : Type*} [Ring A] [StarRing A]

    comm H (star X * X) - star (comm H X) * X - star X * comm H X = 0 := by
  dsimp [comm]
  simp only [star_sub, star_mul, hH]
  noncomm_ring

/-- Purely algebraic Lindblad dissipation identity (zero limits, zero derivatives):
    D_L(X* X) - D_L(X)* X - X* D_L(X) = [L, X]* [L, X]. -/
theorem gksl_dissipator_identity {A : Type*} [Ring A] [StarRing A] [Invertible (2 : A)]
    (L X : A) :
    gkslDissipator L (star X * X) - star (gkslDissipator L X) * X -
      star X * gkslDissipator L X = star (comm L X) * comm L X := by
  rw [star_gkslDissipator]
  dsimp [gkslDissipator, comm]
  rw [mul_sub (star X)]
  have hcommX : star X * (⅟(2 : A) * (star L * L * X + X * star L * L)) =
      ⅟(2 : A) * (star X * (star L * L * X + X * star L * L)) := by
    rw [← mul_assoc, ← invOf_two_comm (star X), mul_assoc]
  have h_two : ⅟(2 : A) + ⅟(2 : A) = 1 := by
    have h2_eq : (2 : A) = 1 + 1 := by ring
    calc ⅟(2 : A) + ⅟(2 : A)
        = ⅟(2 : A) * (1 + 1) := by noncomm_ring
      _ = ⅟(2 : A) * (2 : A) := by rw [← h2_eq]
      _ = 1 := invOf_mul_self (2 : A)
  calc (star L * (star X * X) * L -
        ⅟(2 : A) * (star L * L * (star X * X) + (star X * X) * star L * L)) -

        ⅟(2 : A) * (star L * L * star X + star X * star L * L)) * X -

        star X * (⅟(2 : A) * (star L * L * X + X * star L * L)))

        (star L * star X * L -

         ⅟(2 : A) * (star X * (star L * L * X + X * star L * L))) := by

/-- Conjugation by any unit u ∈ Rˣ as a strict ring endomorphism. -/
def conjRingHom {R : Type*} [Ring R] (u : Rˣ) : R →+* R where
  toFun r := (↑u : R) * r * ↑u⁻¹
  map_one' := by simp
  map_mul' x y := by
    have h : (↑u⁻¹ : R) * ↑u = 1 := u.inv_mul
    calc (↑u : R) * (x * y) * ↑u⁻¹
        = (↑u : R) * x * 1 * y * ↑u⁻¹ := by noncomm_ring
      _ = (↑u : R) * x * ((↑u⁻¹ : R) * ↑u) * y * ↑u⁻¹ := by rw [← h]
      _ = ((↑u : R) * x * ↑u⁻¹) * ((↑u : R) * y * ↑u⁻¹) := by noncomm_ring
  map_zero' := by simp
  map_add' x y := by noncomm_ring

lemma mapMatrix_comm_scalar (K : Type*) {A n : Type*} [CommRing K] [Ring A] [Algebra K A]
    [Fintype n] [DecidableEq n] (W : Matrix n n K) (x : A) :

      Matrix.scalar n x * (algebraMap K A).mapMatrix W := by

theorem shellTwist_comp_scalar (K : Type*) {A n : Type*} [CommRing K] [Ring A] [Algebra K A]
    [Fintype n] [DecidableEq n] (U : (Matrix n n K)ˣ) :
    (conjRingHom (Units.map (algebraMap K A).mapMatrix.toMonoidHom U)).comp (Matrix.scalar n) =
      Matrix.scalar n := by

  have hcomm : (↑U_A : Matrix n n A) * Matrix.scalar n x =
      Matrix.scalar n x * (↑U_A : Matrix n n A) :=
    mapMatrix_comm_scalar K (↑U) x
  have hcancel : (↑U_A : Matrix n n A) * Matrix.scalar n x * ↑U_A⁻¹ = Matrix.scalar n x := by

      _ = Matrix.scalar n x := by rw [U_A.mul_inv, mul_one]
  exact congr_fun (congr_fun hcancel i) j

def iterObj {C : Type*} [Category C] (T : C ⥤ C) (A₀ : C) : ℕ → C
  | 0 => A₀
  | k + 1 => T.obj (iterObj T A₀ k)

def iterStep {C : Type*} [Category C] (T : C ⥤ C) (η : 𝟭 C ⟶ T) (A₀ : C) (k : ℕ) :
    iterObj T A₀ k ⟶ iterObj T A₀ (k + 1) :=
  η.app (iterObj T A₀ k)

def chainMap {C : Type*} [Category C] (X : ℕ → C) (step : ∀ k, X k ⟶ X (k + 1)) :
    ∀ {i j : ℕ}, i ≤ j → (X i ⟶ X j)
  | _, 0, h => eqToHom (by omega)
  | i, j + 1, h =>
    if hij : i ≤ j then chainMap X step hij ≫ step j
    else eqToHom (by omega)

@[simp]
lemma chainMap_refl {C : Type*} [Category C] (X : ℕ → C) (step : ∀ k, X k ⟶ X (k + 1)) (i : ℕ) :
    chainMap X step (le_refl i) = 𝟙 (X i) := by
  cases i with
  | zero => simp [chainMap]
  | succ j =>
    have h : ¬(j + 1 ≤ j) := by omega
    simp [chainMap, h]

lemma chainMap_trans {C : Type*} [Category C] (X : ℕ → C) (step : ∀ k, X k ⟶ X (k + 1))
    {i j k : ℕ} (hij : i ≤ j) (hjk : j ≤ k) :
    chainMap X step (le_trans hij hjk) = chainMap X step hij ≫ chainMap X step hjk := by
  induction k with
  | zero =>
    obtain rfl : i = 0 := by omega
    obtain rfl : j = 0 := by omega
    simp
  | succ m ih =>
    by_cases hjm : j ≤ m
    · have him : i ≤ m := le_trans hij hjm
      simp [chainMap, him, hjm, ih hjm, Category.assoc]
    · obtain rfl : j = m + 1 := by omega
      simp [chainMap_refl]

def inductiveDiagram {C : Type*} [Category C] (T : C ⥤ C) (η : 𝟭 C ⟶ T) (A₀ : C) :
    ℕ ⥤ C where
  obj := iterObj T A₀
  map f := chainMap (iterObj T A₀) (iterStep T η A₀) f.le
  map_id i := chainMap_refl _ _ i
  map_comp f g := chainMap_trans _ _ f.le g.le

/-- Coherent asymptotic symmetry acting on every shell k ∈ ℕ simultaneously. -/
def iterSymmetry {C : Type*} [Category C] (T : C ⥤ C) (τ : T ⟶ T)
    (A₀ : C) (α₀ : A₀ ⟶ A₀) : ∀ k : ℕ, iterObj T A₀ k ⟶ iterObj T A₀ k
  | 0 => α₀
  | k + 1 => T.map (iterSymmetry T τ A₀ α₀ k) ≫ τ.app (iterObj T A₀ k)

lemma iterStep_comm_symmetry {C : Type*} [Category C] (T : C ⥤ C) (η : 𝟭 C ⟶ T)
    (τ : T ⟶ T) (hτ : ∀ X : C, η.app X ≫ τ.app X = η.app X)
    (A₀ : C) (α₀ : A₀ ⟶ A₀) (k : ℕ) :
    iterStep T η A₀ k ≫ iterSymmetry T τ A₀ α₀ (k + 1) =
      iterSymmetry T τ A₀ α₀ k ≫ iterStep T η A₀ k := by
  dsimp [iterStep, iterSymmetry]
  calc η.app _ ≫ (T.map (iterSymmetry T τ A₀ α₀ k) ≫ τ.app _)
      = (η.app _ ≫ T.map (iterSymmetry T τ A₀ α₀ k)) ≫ τ.app _ := by
        simp only [Category.assoc]
    _ = ((𝟭 C).map (iterSymmetry T τ A₀ α₀ k) ≫ η.app _) ≫ τ.app _ := by
        rw [← η.naturality]
    _ = iterSymmetry T τ A₀ α₀ k ≫ (η.app _ ≫ τ.app _) := by
        simp only [Functor.id_map, Category.assoc]
    _ = iterSymmetry T τ A₀ α₀ k ≫ η.app _ := by
        rw [hτ]

lemma chainMap_comm_symmetry {C : Type*} [Category C] (X : ℕ → C)
    (step : ∀ k, X k ⟶ X (k + 1)) (sym : ∀ k, X k ⟶ X k)
    (hcomm : ∀ k, step k ≫ sym (k + 1) = sym k ≫ step k) :
    ∀ {i j : ℕ} (hij : i ≤ j),
      chainMap X step hij ≫ sym j = sym i ≫ chainMap X step hij
  | i, 0, hij => by
    obtain rfl : i = 0 := by omega
    simp [chainMap]
  | i, m + 1, hij => by
    by_cases him : i ≤ m
    · simp only [chainMap, him, ↓reduceDIte]
      rw [Category.assoc, hcomm m, ← Category.assoc,
          chainMap_comm_symmetry X step sym hcomm him, Category.assoc]
    · obtain rfl : i = m + 1 := by omega
      simp [chainMap_refl]

/-- Universal induced asymptotic symmetry α_∞ : colim_k A_k ⟶ colim_k A_k on the colimit algebra,
    satisfying ι_k ≫ α_∞ = α_k ≫ ι_k across all shells k ∈ ℕ without factoring through any finite k₀. -/
def colimitAsymptoticSymmetry {C : Type*} [Category C] (T : C ⥤ C) (η : 𝟭 C ⟶ T)
    (τ : T ⟶ T) (hτ : ∀ X : C, η.app X ≫ τ.app X = η.app X)
    (A₀ : C) (α₀ : A₀ ⟶ A₀)
    (c : Cocone (inductiveDiagram T η A₀)) (hc : IsColimit c) :
    c.pt ⟶ c.pt :=
  hc.desc
    { pt := c.pt
      ι :=
        { app := fun k => iterSymmetry T τ A₀ α₀ k ≫ c.ι.app k
          naturality := fun i j f => by
            dsimp [inductiveDiagram]
            rw [← Category.assoc,
                chainMap_comm_symmetry _ _ _ (iterStep_comm_symmetry T η τ hτ A₀ α₀) f.le,
                Category.assoc]
            exact congr_arg (iterSymmetry T τ A₀ α₀ i ≫ ·) (c.ι.naturality f) } }

end ColimitClosure
