    {Op Pin : Type*} [Ring Op] [Group Pin] [Module ℝ Op]

/--
A chirality-flipping Pin element sends the left numerator to the right
numerator.
-/
theorem rho_mul_one_add_Gamma
    {p : Pin}
    (hp : A.flipsChirality p) :
    A.rho p * ((1 : Op) + A.Gamma) =
      ((1 : Op) - A.Gamma) * A.rho p := by
  calc
    A.rho p * ((1 : Op) + A.Gamma)
        = A.rho p * 1 + A.rho p * A.Gamma := by
            rw [mul_add]
    _ = A.rho p + A.rho p * A.Gamma := by
            simp
    _ = A.rho p + -(A.Gamma * A.rho p) := by
            rw [A.flip_law p hp]
    _ = A.rho p - A.Gamma * A.rho p := by
            simp [sub_eq_add_neg]
    _ = (1 : Op) * A.rho p - A.Gamma * A.rho p := by
            simp
    _ = ((1 : Op) - A.Gamma) * A.rho p := by
            rw [sub_mul]
            simp

A chirality-flipping Pin element sends the right numerator to the left
numerator.
