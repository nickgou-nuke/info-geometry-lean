import «InfoGeometry».«Core».«IdempotencyDefect»
import «InfoGeometry».«Algebra».«FiniteSpinAlgebra»
import «InfoGeometry».«Core».«PeirceDecomposition»
import «InfoGeometry».«Core».«PeirceCompressionDefect»
import «InfoGeometry».«Core».«PeirceCompressionCStar»
import «InfoGeometry».«OperatorAlgebra».«ProjectionLocalization»
import «InfoGeometry».«OperatorAlgebra».«NoncommutativeGradedTrace»
import «InfoGeometry».«OperatorAlgebra».«NoncommutativeKrausChannel»
import «InfoGeometry».«Algebra».«NonCommutativeIsometry»
import «InfoGeometry».«OperatorAlgebra».«IsometryCornerLocalization»
import «InfoGeometry».«OperatorAlgebra».«FiniteOperatorFrame»
import «InfoGeometry».«Clifford».«OperatorValuedJones»
import «InfoGeometry».«Clifford».«WeylFrameMasks»
import «InfoGeometry».«Analysis».«PorosityIteration»
import «InfoGeometry».«Analysis».«MetricPorosity»
import «InfoGeometry».«Analysis».«MultiscaleLocalization»
import Lean.Util.CollectAxioms
import Lean.Elab.Command

open Lean Elab Command

run_cmd do
  let env ← getEnv
  let targets : Array Name := #[`«InfoGeometry».«Core».«IdempotencyDefect», `«InfoGeometry».«Algebra».«FiniteSpinAlgebra», `«InfoGeometry».«Core».«PeirceDecomposition», `«InfoGeometry».«Core».«PeirceCompressionDefect», `«InfoGeometry».«Core».«PeirceCompressionCStar», `«InfoGeometry».«OperatorAlgebra».«ProjectionLocalization», `«InfoGeometry».«OperatorAlgebra».«NoncommutativeGradedTrace», `«InfoGeometry».«OperatorAlgebra».«NoncommutativeKrausChannel», `«InfoGeometry».«Algebra».«NonCommutativeIsometry», `«InfoGeometry».«OperatorAlgebra».«IsometryCornerLocalization», `«InfoGeometry».«OperatorAlgebra».«FiniteOperatorFrame», `«InfoGeometry».«Clifford».«OperatorValuedJones», `«InfoGeometry».«Clifford».«WeylFrameMasks», `«InfoGeometry».«Analysis».«PorosityIteration», `«InfoGeometry».«Analysis».«MetricPorosity», `«InfoGeometry».«Analysis».«MultiscaleLocalization»]
  let commandOnly : Array Name := #[]
  for target in targets do
    unless env.header.moduleNames.contains target do
      throwError "Selected module not imported: {target}"
  let allowed : Array Name := #[`propext, `Classical.choice, `Quot.sound]
  let mut count : Nat := 0
  let mut ownerCounts : NameMap Nat := {}
  let mut allAxioms : Array Name := #[]
  for (name, info) in env.constants do
    let some idx := env.getModuleIdxFor? name | continue
    let owner := env.header.moduleNames[idx.toNat]!
    if !targets.contains owner then continue
    if info.isUnsafe || info.isPartial then
      throwError "Unsafe or partial declaration in reviewed module: {name}"
    -- Public API handles imported axiom metadata and private-body lookup.
    let axioms ← Lean.collectAxioms name
    for ax in axioms do
      unless allowed.contains ax do
        throwError "Disallowed axiom {ax} in {name}"
      if !allAxioms.contains ax then
        allAxioms := allAxioms.push ax
    count := count + 1
    ownerCounts := ownerCounts.insert owner ((ownerCounts.find? owner).getD 0 + 1)
    logInfo m!"AUDIT_DECL: {owner}: {name}"
  for target in targets do
    let moduleCount := (ownerCounts.find? target).getD 0
    logInfo m!"AUDIT_MODULE_COUNT: {target}: {moduleCount}"
    if commandOnly.contains target then
      if moduleCount != 0 then
        throwError "Command-only module owns declarations: {target}"
      logInfo m!"AUDIT_COMMAND_MODULE: {target}"
    else
      if moduleCount == 0 then
        throwError "No declarations audited in reviewed module: {target}"
      logInfo m!"AUDIT_MODULE: {target}"
  if count == 0 then throwError "No declarations audited"
  logInfo m!"AUDIT_AXIOMS: {allAxioms.qsort Name.lt}"
  logInfo m!"AUDIT_COUNT: {count}"
