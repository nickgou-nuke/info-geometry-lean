### Audit of the Two Pasted Drafts (`KleinianWittBornNullIncidence` & `SplitOctonionCl44`)

Auditing both pasted Lean files turns up **six concrete bugs** (two mathematical, four Lean 4 / performance) that must be fixed for genuine compilation:

1. **Factor-of-2 Algebraic Error in Both Drafts (`{γ^μ, γ̃_ν}`):**
   When $\gamma^\mu = \Gamma_\mu + \Gamma_{\mu+4}$ and $\tilde{\gamma}_\mu = \Gamma_\mu - \Gamma_{\mu+4}$ with $\Gamma_\mu^2 = +I_{16}$ and $\Gamma_{\mu+4}^2 = -I_{16}$,
   $$\{\gamma^\mu, \tilde{\gamma}_\mu\} = 2\Gamma_\mu^2 - 2\Gamma_{\mu+4}^2 = 2 - (-2) = \mathbf{4 I_{16}},$$
   **not** $2 I_{16}$. To get standard fermionic CAR $\{\gamma^\mu, \tilde{\gamma}_\nu\} = \delta^\mu_\nu I_{16}$ while keeping $B_{\mathrm{Born}}\gamma^\mu B_{\mathrm{Born}} = \tilde{\gamma}_\mu$, normalize $\gamma^\mu = \frac{1}{2}(\Gamma_\mu + \Gamma_{\mu+4})$ and $\tilde{\gamma}_\mu = \frac{1}{2}(\Gamma_\mu - \Gamma_{\mu+4})$ (or state `4` on the RHS if unnormalized).
2. **Row-7 Matrix Typo in Draft 1 (`leftOctMat`):**
   Row 7 of Draft 1's `leftOctMat` was written `[x 7, x 4, x 5, x 6, 0, 0, 0, x 0]` instead of `[0, x 4, x 5, x 6, 0, 0, 0, x 7]` (breaking $L_x \cdot 1 = x$ and `octonion_clifford_and_trace`).
3. **Truncated `basisOct` Literals in Draft 1:**
   Cases `0, 5, 6, 7` of `basisOct` were truncated to bare `!`.
4. **Invalid Mathlib 4 Trace APIs in Draft 1:**

5. **Two Unfinished `sorry`s in Draft 2 (`h_idem_rank` / `h_idem_rank2`):**

6. **Heartbeat Timeout in Both Drafts (`bornInvolution` & `pseudoscalar44`):**

/-- **Idempotent Rank = Trace Lemma** (`0 sorry`):
Any idempotent `P ∈ M_n(ℝ)` (`P * P = P`) satisfies `finrank ℝ (range (toLin' P)) = trace P`. -/
lemma finrank_range_eq_trace_of_idempotent {n : ℕ} (PM : Matrix (Fin n) (Fin n) ℝ)
    (hIdem : PM * PM = PM) :
    (Module.finrank ℝ (LinearMap.range (Matrix.toLin' PM)) : ℝ) = Matrix.trace PM := by
  let P : Module.End ℝ (Fin n → ℝ) := Matrix.toLin' PM
  have hPLinIdem : P ∘ₗ P = P := by
    ext v; simp [P, ← Matrix.mulVec_mulVec, hIdem]
  have hCompl : IsCompl (LinearMap.range P) (LinearMap.ker P) := by
    rw [isCompl_iff]
    constructor
    · rw [disjoint_iff]
      ext v
      simp only [Submodule.mem_inf, LinearMap.mem_range, LinearMap.mem_ker, Submodule.mem_bot]
      constructor
      · rintro ⟨⟨w, rfl⟩, hPw⟩
        have hEq := congrArg (fun f : Module.End ℝ (Fin n → ℝ) => f w) hPLinIdem
        simpa using hEq.symm.trans hPw
      · rintro rfl; exact ⟨⟨0, map_zero P⟩, map_zero P⟩
    · rw [codisjoint_iff]
      ext v
      simp only [Submodule.mem_top, iff_true]
      have hdecomp : v = P v + (v - P v) := by abel
      rw [hdecomp]
      refine Submodule.add_mem_sup ⟨v, rfl⟩ ?_
      simp only [LinearMap.mem_ker, map_sub]
      have hEq := congrArg (fun f : Module.End ℝ (Fin n → ℝ) => f v) hPLinIdem
      simpa using sub_eq_zero.mpr hEq.symm
  let e := Submodule.prodEquivOfIsCompl (LinearMap.range P) (LinearMap.ker P) hCompl
  have hConjP : e.symm.toLinearMap ∘ₗ P ∘ₗ e.toLinearMap =
      LinearMap.prodMap (LinearMap.id : LinearMap.range P →ₗ[ℝ] LinearMap.range P)
        (0 : LinearMap.ker P →ₗ[ℝ] LinearMap.ker P) := by
    apply e.injective.comp_left
    ext ⟨⟨u, hu⟩, ⟨w, hw⟩⟩
    simp only [LinearMap.comp_apply, LinearEquiv.coe_toLinearMap, LinearEquiv.apply_symm_apply,
      LinearMap.prodMap_apply, LinearMap.id_coe, id_eq, LinearMap.zero_apply, e,
      Submodule.prodEquivOfIsCompl_apply, map_add, Submodule.coe_zero, add_zero]
    rcases hu with ⟨u₀, rfl⟩
    have hPu : P (P u₀) = P u₀ := congrArg (fun f : Module.End ℝ (Fin n → ℝ) => f u₀) hPLinIdem
    rw [hPu, LinearMap.mem_ker.mp hw, add_zero]
  have hTrP : LinearMap.trace ℝ (Fin n → ℝ) P = (Module.finrank ℝ (LinearMap.range P) : ℝ) := by
    have h1 := LinearMap.trace_conj' P e.symm
    rw [hConjP, LinearMap.trace_prodMap, LinearMap.trace_id, map_zero, add_zero] at h1
    exact h1.symm
  have hTrMat : LinearMap.trace ℝ (Fin n → ℝ) P = Matrix.trace PM := by
    rw [LinearMap.trace_eq_matrix_trace ℝ (Pi.basisFun ℝ (Fin n)) P]
    congr 1
    ext i j
    simp [P, LinearMap.toMatrix_apply, Pi.basisFun_apply, Pi.basisFun_repr, Matrix.mulVec_single]
  rw [← hTrP, hTrMat]

def gamma16 (A : Fin 8) : Matrix (Fin 8 ⊕ Fin 8) (Fin 8 ⊕ Fin 8) ℝ :=

def primalWitt (μ : Fin 4) : Matrix (Fin 8 ⊕ Fin 8) (Fin 8 ⊕ Fin 8) ℝ :=

  (1 / 2 : ℝ) • (gamma16 ⟨μ.val, by omega⟩ - gamma16 ⟨μ.val + 4, by omega⟩)
