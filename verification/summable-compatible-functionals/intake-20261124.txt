### 1. Точното съответствие с техния `EffectiveForm.lean` и `Convolution.lean`

В техния код решетъчното скаларно произведение въвежда обемния множител $l^{-d}$:
- В `EffectiveForm.lean`:
  $$\langle x,\; \mathrm{blockAdjoint}\,\psi \rangle_{\mathrm{fine}} = l^{-d}\,\langle \mathrm{blockAve}\,x,\; \psi \rangle_{\mathrm{coarse}},$$
  т.е. спрямо стандартното евклидово транспониране $Q^T = (\mathrm{blockAve})^T$ техният оператор е:
  $$\mathrm{blockAdjoint} = l^{-d} Q^T, \qquad \mathrm{blockMatrix} = \mathrm{blockAdjoint} \circ \mathrm{blockAve} = l^{-d} Q^T Q.$$
- Когато в нашия `PositiveCoarseGraining` положим $Q = \mathrm{blockAve}$ и $R_{\mathrm{lat}} = (\alpha l^{-d}) R$:
  1. **`effectiveHessian α H`**:
     $$H_{\mathrm{eff}} = A + Q^T R_{\mathrm{lat}} Q = H + \alpha l^{-d} Q^T R Q \xrightarrow{R = I} H + (\alpha l^{-d})\,\mathrm{blockMatrix}.$$
  2. **`minimizerField α H ψ`**:
     $$E \psi = H_{\mathrm{eff}}^{-1} Q^T R_{\mathrm{lat}} \psi = \alpha H_{\mathrm{eff}}^{-1} (\mathrm{blockAdjoint}\, R \psi) \xrightarrow{R = I} \alpha H_{\mathrm{eff}}^{-1}(\mathrm{blockAdjoint}\,\psi).$$
  3. **`effectiveOperator α H`**:
     $$S = R_{\mathrm{lat}} - R_{\mathrm{lat}} Q H_{\mathrm{eff}}^{-1} Q^T R_{\mathrm{lat}} = l^{-d} \alpha \bigl(R - \alpha R (\mathrm{blockAve}) H_{\mathrm{eff}}^{-1} (\mathrm{blockAdjoint}) R\bigr).$$
  4. **Специализацията в `Convolution.lean` (`integral_blockKernel_mul_blockKernel`)**:
     В техния код `blockAve_comp_blockAdjoint` е $\mathrm{blockAve} \circ \mathrm{blockAdjoint} = l^d I$, т.е. $Q_2 Q_2^T = l^{2d} I$. При $s = 1$ (след нормировка с $l^{-d}$) общата формула $(R_2^{-1} + Q_2 R_1^{-1} Q_2^T)^{-1}$ при $R_1 = \beta I$, $R_2 = \alpha I$, $Q_2 Q_2^T = s I$ дава:
     $$R_{\mathrm{eff}} = \frac{\alpha\beta}{\beta + s\alpha} I \xrightarrow{s = 1} \frac{\alpha\beta}{\beta + \alpha} I,$$
     което съвпада буквално с `γ := α * β / (β + α)` в `Convolution.lean` (ред 108).

### 2. Готов Lean 4 код за Първия PR: Специализация на $R_{\mathrm{eff}}$, Решетъчна Скалировка и Некомутиращ Пример

Този модул се компилира директно върху нашия `CoarseGrainingComposition.lean` и доказва както скаларната специализация за `Convolution.lean`, така и явен $2 \times 2$ некомутиращ свидетел ($R_1 Q_2^T \neq Q_2^T R_1$ и $R_1 R_2 \neq R_2 R_1$):

```lean
/-

Released under Apache 2.0 license as described in the file LICENSE.
Authors: InfoGeometry Engine Contributors

import Mathlib.Data.Matrix.Basic
import Mathlib.Tactic

noncomputable section

open Matrix

namespace BalabanRGAdapter

/-! ### 1. Specialization of General Matrix Composition to `Convolution.lean` Scalar Formula -/

/-- Under `R₁ = β • I`, `R₂ = α • I`, and `Q₂ * Q₂ᵀ = s • I`, the two-step effective noise
covariance `R_eff⁻¹ = R₂⁻¹ + Q₂ R₁⁻¹ Q₂ᵀ` has exact inverse
`R_eff = (α * β / (β + s * α)) • I`, recovering `γ = α * β / (β + α)` at `s = 1`. -/
theorem scalar_blockKernel_composition_specialization
    {n m : Type*} [Fintype n] [DecidableEq n] [Fintype m] [DecidableEq m]
    (Q₂ : Matrix m n ℝ) (α β s : ℝ)
    (hα : 0 < α) (hβ : 0 < β) (hs : 0 ≤ s)
    (hQQ : Q₂ * Q₂ᵀ = s • (1 : Matrix m m ℝ)) :
    let R₁_inv : Matrix n n ℝ := β⁻¹ • 1
    let R₂_inv : Matrix m m ℝ := α⁻¹ • 1
    let R_eff_inv : Matrix m m ℝ := R₂_inv + Q₂ * R₁_inv * Q₂ᵀ
    let γ : ℝ := (α * β) / (β + s * α)
    R_eff_inv = ((β + s * α) / (α * β)) • (1 : Matrix m m ℝ) ∧
    R_eff_inv * (γ • (1 : Matrix m m ℝ)) = 1 ∧
    (s = 1 → γ = (α * β) / (β + α)) := by
  dsimp
  have hαNe : α ≠ 0 := ne_of_gt hα
  have hβNe : β ≠ 0 := ne_of_gt hβ
  have hDenPos : 0 < β + s * α := by nlinarith
  have hDenNe : β + s * α ≠ 0 := ne_of_gt hDenPos
  have hId : α⁻¹ • (1 : Matrix m m ℝ) + Q₂ * (β⁻¹ • (1 : Matrix n n ℝ)) * Q₂ᵀ =
      ((β + s * α) / (α * β)) • (1 : Matrix m m ℝ) := by
    rw [Matrix.mul_smul, Matrix.mul_one, Matrix.smul_mul, hQQ, smul_smul, ← add_smul]
    congr 1
    field_simp [hαNe, hβNe]
    ring
  refine ⟨hId, ?_, fun hs1 => by rw [hs1, one_mul]⟩
  rw [hId, smul_mul_assoc, one_mul, mul_smul_comm, mul_one, smul_smul]
  have hCancel : (β + s * α) / (α * β) * (α * β / (β + s * α)) = 1 := by
    field_simp [hαNe, hβNe, hDenNe]
  rw [hCancel, one_smul]

/-! ### 2. Lattice-Weighted Adjoint Adapter (`blockAdjoint = l⁻ᵈ • Qᵀ`) -/

/-- Reconciles `EffectiveForm.lean`'s lattice-scaled `blockAdjoint = w • Qᵀ` (`w = l⁻ᵈ`) and
`blockMatrix = blockAdjoint * Q = w • (Qᵀ * Q)` with the matrix-weighted coarse precision
`R_lat = (α * w) • R`: at `R = 1`, `A + Qᵀ * R_lat * Q` equals `A + (α * w) • blockMatrix`. -/
theorem lattice_weighted_effectiveHessian_bridge
    {n m : Type*} [Fintype n] [DecidableEq n] [Fintype m] [DecidableEq m]
    (A : Matrix n n ℝ) (Q : Matrix m n ℝ) (R : Matrix m m ℝ) (α w : ℝ) :
    let blockAdjoint : Matrix n m ℝ := w • Qᵀ
    let blockMatrix : Matrix n n ℝ := blockAdjoint * Q
    let R_lat : Matrix m m ℝ := (α * w) • R
    (A + Qᵀ * R_lat * Q = A + α • (blockAdjoint * R * Q)) ∧
    (R = 1 → A + Qᵀ * R_lat * Q = A + (α * w) • blockMatrix) := by
  dsimp
  constructor
  · simp only [Matrix.mul_smul, Matrix.smul_mul, smul_smul]
    congr 1
    rw [mul_comm α w]
  · rintro rfl
    simp only [Matrix.mul_smul, Matrix.smul_mul, Matrix.mul_one]

/-! ### 3. Explicit Non-Commuting 2×2 Matrix Witness (Section 6, Item 5) -/

def R1_nc : Matrix (Fin 2) (Fin 2) ℝ := !![2, 1; 1, 1]
def R2_nc : Matrix (Fin 2) (Fin 2) ℝ := !![1, 0; 0, 3]
def Q2_nc : Matrix (Fin 2) (Fin 2) ℝ := !![1, 1; 0, 1]

/-- **Non-Commuting 2×2 Coarse-Graining Witness**:
`R1_nc` and `R2_nc` are symmetric positive-definite matrices that **do not commute**
(`R1_nc * R2_nc ≠ R2_nc * R1_nc`), and `Q2_nc` does not commute with `R1_nc`, verifying that
ordered sandwiching `Q₂ * R₁⁻¹ * Q₂ᵀ` is essential and strictly differs from `R₁⁻¹ * Q₂ * Q₂ᵀ`. -/
theorem noncommuting_coarse_graining_witness :
    (R1_ncᵀ = R1_nc ∧ R2_ncᵀ = R2_nc) ∧
    (R1_nc * R2_nc ≠ R2_nc * R1_nc) ∧
    (Q2_nc * R1_nc * Q2_ncᵀ ≠ R1_nc * (Q2_nc * Q2_ncᵀ)) := by
  refine ⟨⟨?_, ?_⟩, ?_, ?_⟩
  · ext i j; fin_cases i <;> fin_cases j <;> rfl
  · ext i j; fin_cases i <;> fin_cases j <;> rfl
  · intro hEq
    have h01 := congrFun (congrFun hEq 0) 1
    simp [R1_nc, R2_nc, Matrix.mul_apply, Fin.sum_univ_two] at h01
  · intro hEq
    have h10 := congrFun (congrFun hEq 1) 0
    simp [R1_nc, Q2_nc, Matrix.mul_apply, Fin.sum_univ_two] at h10

end BalabanRGAdapter
```

### 3. Новата теорема от Раздел 4: Приблизителна съвместимост със сумируема грешка $\implies$ Точно съвместими гранични състояния

Ето пълната формализация в Lean 4 на предложената от теб теорема в Раздел 4: ако на всяка стъпка $n$ имаме контрактилно влагане $\|\iota_n a\| \le \|a\|$ и приблизителна съвместимост на състоянията
$$\lvert \omega_{n+1}(\iota_n a) - \omega_n(a) \rvert \le \varepsilon_n \|a\|, \qquad \sum_{n=0}^\infty \varepsilon_n < \infty,$$
то редицата $m \mapsto \omega_{n+m}(\iota_{n, n+m} a)$ е редица на Коши в $\mathbb{R}$, сходи се към гранично състояние $\omega_n^\infty(a)$, удовлетворява равномерната оценка
$$\lvert \omega_n^\infty(a) - \omega_n(a) \rvert \le \left(\sum_{k=0}^\infty \varepsilon_{n+k}\right) \|a\|,$$
и **граничните състояния са точно проективно съвместими**:
$$\boxed{\omega_{n+1}^\infty(\iota_n a) = \omega_n^\infty(a)}$$
— което дава точния вход `ω (n + 1) (ι n a) = ω n a` за нашия `CStarAlgebraStateColimit.lean`!

```lean
/-
Copyright (c) 2026 InfoGeometry Authors. All rights reserved.
Released under Apache 2.0 license as described in the file LICENSE.
Authors: InfoGeometry Engine Contributors
-/
import Mathlib.Analysis.Normed.Group.Basic
import Mathlib.Topology.Algebra.InfiniteSum.Real
import Mathlib.Topology.Instances.Real
import Mathlib.Tactic

noncomputable section

open scoped BigOperators
open Filter

namespace ApproximateCompatibleStateColimit

/-- Directed multi-step refinement `ιSteps ι n m : A n → A (n + m)` defined inductively on `m`. -/
def ιSteps {A : ℕ → Type*} (ι : ∀ n, A n → A (n + 1)) :
    (n m : ℕ) → A n → A (n + m)
  | _, 0, a => a
  | n, m + 1, a =>
      cast (by congr 1; omega) (ι (n + m) (ιSteps ι n m a))

/-- Explicit step-by-step sequence `seqApprox ι ω n a m = ω (n + m) (ι_{n, n+m} a)` by recursion
on `m` (avoiding `cast` friction!). -/
def seqApprox {A : ℕ → Type*} (ι : ∀ n, A n → A (n + 1)) (ω : ∀ n, A n → ℝ) :
    (n : ℕ) → A n → ℕ → ℝ
  | n, a, 0 => ω n a
  | n, a, m + 1 => seqApprox ι ω (n + 1) (ι n a) m

/-- Fundamental one-step shift identity: `seqApprox ι ω n a (m + 1) = seqApprox ι ω (n + 1) (ι n a) m`. -/
@[simp]
lemma seqApprox_succ {A : ℕ → Type*} (ι : ∀ n, A n → A (n + 1)) (ω : ∀ n, A n → ℝ)
    (n : ℕ) (a : A n) (m : ℕ) :
    seqApprox ι ω n a (m + 1) = seqApprox ι ω (n + 1) (ι n a) m := rfl

/-- Consecutive difference bound:
`|seqApprox ι ω n a (m + 1) - seqApprox ι ω n a m| ≤ ε (n + m) * ‖a‖`
whenever `ι` is contractive (`‖ι n a‖ ≤ ‖a‖`) and `|ω (n+1) (ι n a) - ω n a| ≤ ε n * ‖a‖`. -/
lemma abs_seqApprox_succ_sub_le
    {A : ℕ → Type*} [∀ n, SeminormedAddCommGroup (A n)]
    (ι : ∀ n, A n → A (n + 1)) (ω : ∀ n, A n → ℝ) (ε : ℕ → ℝ)
    (hε_nn : ∀ n, 0 ≤ ε n)
    (hContr : ∀ n (a : A n), ‖ι n a‖ ≤ ‖a‖)
    (hApprox : ∀ n (a : A n), |ω (n + 1) (ι n a) - ω n a| ≤ ε n * ‖a‖) :
    ∀ (m n : ℕ) (a : A n),
      |seqApprox ι ω n a (m + 1) - seqApprox ι ω n a m| ≤ ε (n + m) * ‖a‖ := by
  intro m
  induction m with
  | zero =>
    intro n a
    simpa [seqApprox] using hApprox n a
  | succ k ih =>
    intro n a
    calc |seqApprox ι ω n a (k + 2) - seqApprox ι ω n a (k + 1)|
        = |seqApprox ι ω (n + 1) (ι n a) (k + 1) - seqApprox ι ω (n + 1) (ι n a) k| := rfl
      _ ≤ ε (n + 1 + k) * ‖ι n a‖ := ih (n + 1) (ι n a)
      _ ≤ ε (n + 1 + k) * ‖a‖ :=
          mul_le_mul_of_nonneg_left (hContr n a) (hε_nn _)
      _ = ε (n + (k + 1)) * ‖a‖ := by congr 2; omega

/-- Telescoping finite-horizon bound:
`|seqApprox ι ω n a m - ω n a| ≤ (∑ k ∈ range m, ε (n + k)) * ‖a‖`. -/
lemma abs_seqApprox_sub_initial_le
    {A : ℕ → Type*} [∀ n, SeminormedAddCommGroup (A n)]
    (ι : ∀ n, A n → A (n + 1)) (ω : ∀ n, A n → ℝ) (ε : ℕ → ℝ)
    (hε_nn : ∀ n, 0 ≤ ε n)
    (hContr : ∀ n (a : A n), ‖ι n a‖ ≤ ‖a‖)
    (hApprox : ∀ n (a : A n), |ω (n + 1) (ι n a) - ω n a| ≤ ε n * ‖a‖)
    (n : ℕ) (a : A n) (m : ℕ) :
    |seqApprox ι ω n a m - ω n a| ≤ (∑ k ∈ Finset.range m, ε (n + k)) * ‖a‖ := by
  induction m with
  | zero => simp [seqApprox]
  | succ k ih =>
    rw [Finset.sum_range_succ, add_mul]
    have hTri : |seqApprox ι ω n a (k + 1) - ω n a| ≤
        |seqApprox ι ω n a k - ω n a| +
        |seqApprox ι ω n a (k + 1) - seqApprox ι ω n a k| := by
      have hEq : seqApprox ι ω n a (k + 1) - ω n a =
          (seqApprox ι ω n a k - ω n a) +
          (seqApprox ι ω n a (k + 1) - seqApprox ι ω n a k) := by ring
      rw [hEq]
      exact abs_add_le _ _
    exact hTri.trans (add_le_add ih
      (abs_seqApprox_succ_sub_le ι ω ε hε_nn hContr hApprox k n a))

/-- **Main Theorem (Summable Approximate RG Compatibility Implies Exact Limit State Compatibility)**:
If `|ω_{n+1}(ι_n a) - ω_n(a)| ≤ ε_n ‖a‖` with `∑ ε_n < ∞` and `‖ι_n a‖ ≤ ‖a‖`:
1. **Convergence**: For every `n` and `a : A n`, `m ↦ seqApprox ι ω n a m` converges to a limit
   `ω_∞ n a : ℝ`.
2. **Exact Projective State Compatibility**:
   $$\omega_\infty(n + 1)(\iota_n a) = \omega_\infty(n)(a),$$
   providing the exact compatibility hypothesis required by `CStarAlgebraStateColimit`.
3. **Quantitative Tail Error Bound**:
   $$|\omega_\infty(n)(a) - \omega_n(a)| \le \left(\sum_{k=0}^\infty \varepsilon_{n+k}\right) \|a\|.$$ -/
theorem summable_approx_states_converge_to_exact_compatible_state
    {A : ℕ → Type*} [∀ n, SeminormedAddCommGroup (A n)]
    (ι : ∀ n, A n → A (n + 1)) (ω : ∀ n, A n → ℝ) (ε : ℕ → ℝ)
    (hε_nn : ∀ n, 0 ≤ ε n) (hSum : Summable ε)
    (hContr : ∀ n (a : A n), ‖ι n a‖ ≤ ‖a‖)
    (hApprox : ∀ n (a : A n), |ω (n + 1) (ι n a) - ω n a| ≤ ε n * ‖a‖) :
    ∃ ω_inf : ∀ n, A n → ℝ,
      (∀ n (a : A n), Tendsto (seqApprox ι ω n a) atTop (nhds (ω_inf n a))) ∧
      (∀ n (a : A n), ω_inf (n + 1) (ι n a) = ω_inf n a) ∧
      (∀ n (a : A n), |ω_inf n a - ω n a| ≤ (∑' k : ℕ, ε (n + k)) * ‖a‖) := by
  have hCauchy : ∀ n (a : A n), CauchySeq (seqApprox ι ω n a) := by
    intro n a
    apply cauchySeq_of_summable_dist
    have hDistLe : ∀ m : ℕ,
        dist (seqApprox ι ω n a m) (seqApprox ι ω n a (m + 1)) ≤ ε (n + m) * ‖a‖ := by
      intro m
      rw [dist_comm, Real.dist_eq]
      exact abs_seqApprox_succ_sub_le ι ω ε hε_nn hContr hApprox m n a
    have hSumShift : Summable (fun m : ℕ => ε (n + m) * ‖a‖) := by
      have h1 : Summable (fun m : ℕ => ε (m + n)) := (summable_nat_add_iff n).mpr hSum
      have h2 : (fun m : ℕ => ε (n + m)) = (fun m : ℕ => ε (m + n)) := by
        ext m; rw [add_comm]
      rw [h2]
      exact h1.mul_right ‖a‖
    exact Summable.of_nonneg_of_le (fun _ => dist_nonneg) hDistLe hSumShift
  choose ω_inf hTendsto using fun n (a : A n) => cauchySeq_tendsto_of_complete (hCauchy n a)
  refine ⟨ω_inf, hTendsto, ?_, ?_⟩
  · intro n a
    have hShift : Tendsto (fun m => seqApprox ι ω n a (m + 1)) atTop (nhds (ω_inf n a)) :=
      (tendsto_add_atTop_iff_nat 1).mpr (hTendsto n a)
    exact tendsto_nhds_unique (hTendsto (n + 1) (ι n a)) hShift
  · intro n a
    have hSumShift : Summable (fun k : ℕ => ε (n + k)) := by
      have h1 : Summable (fun k : ℕ => ε (k + n)) := (summable_nat_add_iff n).mpr hSum
      have h2 : (fun k : ℕ => ε (n + k)) = (fun k : ℕ => ε (k + n)) := by
        ext k; rw [add_comm]
      rwa [h2]
    have hLHS : Tendsto (fun m => |seqApprox ι ω n a m - ω n a|) atTop
        (nhds |ω_inf n a - ω n a|) :=
      ((hTendsto n a).sub tendsto_const_nhds).abs
    have hRHS : Tendsto (fun m => (∑ k ∈ Finset.range m, ε (n + k)) * ‖a‖) atTop
        (nhds ((∑' k : ℕ, ε (n + k)) * ‖a‖)) :=
      (hSumShift.hasSum.tendsto_sum_nat).mul_const ‖a‖
    exact le_of_tendsto_of_tendsto hLHS hRHS
      (Eventually.of_forall (abs_seqApprox_sub_initial_le ι ω ε hε_nn hContr hApprox n a))

end ApproximateCompatibleStateColimit
```
