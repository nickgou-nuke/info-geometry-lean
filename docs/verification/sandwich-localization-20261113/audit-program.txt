import InfoGeometry.Algebra.SandwichLocalization
import InfoGeometry.Algebra.SandwichLocalizationActions
import InfoGeometry.Algebra.SandwichLocalizationStar
import InfoGeometry.Algebra.SandwichBalancedTensor
import InfoGeometry.Algebra.OneSidedLocalizationBoundary
import InfoGeometry.OperatorAlgebra.PeirceSpine.SandwichCayley
import InfoGeometry.OperatorAlgebra.PeirceSpine.EndogenousKernelBridge
import InfoGeometry.OperatorAlgebra.PeirceSpine.PeirceDefectIdentities
import InfoGeometry.OperatorAlgebra.PeirceSpine.KernelBridgeMatrixWitness
import Lean.Util.CollectAxioms
import Lean.Elab.Command

open Lean Elab Command

run_cmd do
  let env ← getEnv
  let targets : Array Name := #[`InfoGeometry.Algebra.SandwichLocalization, `InfoGeometry.Algebra.SandwichLocalizationActions, `InfoGeometry.Algebra.SandwichLocalizationStar, `InfoGeometry.Algebra.SandwichBalancedTensor, `InfoGeometry.Algebra.OneSidedLocalizationBoundary, `InfoGeometry.OperatorAlgebra.PeirceSpine.SandwichCayley, `InfoGeometry.OperatorAlgebra.PeirceSpine.EndogenousKernelBridge, `InfoGeometry.OperatorAlgebra.PeirceSpine.PeirceDefectIdentities, `InfoGeometry.OperatorAlgebra.PeirceSpine.KernelBridgeMatrixWitness]
  let allowed : Array Name := #[`propext, `Classical.choice, `Quot.sound]
  let mut count : Nat := 0
  let mut allAxioms : Array Name := #[]
  for (name, info) in env.constants do
    let some idx := env.getModuleIdxFor? name | continue
    let owner := env.header.moduleNames[idx.toNat]!
    if !targets.contains owner then continue
    if info.isUnsafe || info.isPartial then
      throwError "Unsafe or partial declaration in reviewed module: {name}"
    let axioms ← Lean.collectAxioms name
    for ax in axioms do
      unless allowed.contains ax do
        throwError "Disallowed axiom {ax} in {name}"
      unless allAxioms.contains ax do
        allAxioms := allAxioms.push ax
    count := count + 1
    logInfo m!"AUDIT_DECL: {owner}: {name}: {axioms}"
  if count == 0 then throwError "No declarations audited"
  logInfo m!"AUDIT_AXIOMS: {allAxioms}"
  logInfo m!"AUDIT_COUNT: {count}"
