   `D₁₂(v ⋆ s) = D₁₂(v) ⋆ s + v ⋆ D₁₂(s)` with `D₁₂(e₁) = -e₂ ≠ 0`. -/
theorem cartan_infinitesimal_triality_and_g22_derivation (a b v s : ZornOct ℝ) :
    (spinBivPlus a b (zMul v s) =
      zMul (vecBiv a b v) s + zMul v (spinBivMinus a b s)) ∧
    (let D_plus := spinBivPlus (quatE 0) (quatE 1) - spinBivPlus (splitEllE 0) (splitEllE 1)
     let D_minus := spinBivMinus (quatE 0) (quatE 1) - spinBivMinus (splitEllE 0) (splitEllE 1)
     let D_vec := fun x => vecBiv (quatE 0) (quatE 1) x - vecBiv (splitEllE 0) (splitEllE 1) x
     D_plus v = D_vec v ∧
     D_minus v = D_vec v ∧
     D_vec (zMul v s) = zMul (D_vec v) s + zMul v (D_vec s) ∧
     D_vec (quatE 0) = -quatE 1) := by
  have hTri : spinBivPlus a b (zMul v s) =
      zMul (vecBiv a b v) s + zMul v (spinBivMinus a b s) := by
    have hCl := (cl44_real_16d_spinor_representation a v 0).1
    have hCl_bv := (cl44_real_16d_spinor_representation b v 0).1
    have hComm :
        ((1 / 4 : ℝ) • (gamma44 a * gamma44 b - gamma44 b * gamma44 a)) * gamma44 v -
        gamma44 v * ((1 / 4 : ℝ) • (gamma44 a * gamma44 b - gamma44 b * gamma44 a)) =
        zBilin b v • gamma44 a - zBilin a v • gamma44 b := by
      have h1 : gamma44 a * gamma44 b * gamma44 v - gamma44 v * (gamma44 a * gamma44 b) =
          (2 * zBilin b v) • gamma44 a - (2 * zBilin a v) • gamma44 b := by
        have hbv : gamma44 b * gamma44 v =
            (2 * zBilin b v) • (1 : Module.End ℝ Spinor16) - gamma44 v * gamma44 b :=
          eq_sub_of_add_eq hCl_bv
        have hav : gamma44 a * gamma44 v =
            (2 * zBilin a v) • (1 : Module.End ℝ Spinor16) - gamma44 v * gamma44 a :=
          eq_sub_of_add_eq hCl
        calc gamma44 a * gamma44 b * gamma44 v - gamma44 v * (gamma44 a * gamma44 b)
            = gamma44 a * (gamma44 b * gamma44 v) - gamma44 v * gamma44 a * gamma44 b := by
                simp only [mul_assoc]
          _ = gamma44 a * ((2 * zBilin b v) • 1 - gamma44 v * gamma44 b) -
                gamma44 v * gamma44 a * gamma44 b := by rw [hbv]
          _ = (2 * zBilin b v) • gamma44 a - (gamma44 a * gamma44 v) * gamma44 b -
                gamma44 v * gamma44 a * gamma44 b := by
                rw [mul_sub, mul_smul_comm, mul_one]; simp only [mul_assoc]
          _ = (2 * zBilin b v) • gamma44 a -
                ((2 * zBilin a v) • 1 - gamma44 v * gamma44 a) * gamma44 b -
                gamma44 v * gamma44 a * gamma44 b := by rw [hav]
          _ = (2 * zBilin b v) • gamma44 a - (2 * zBilin a v) • gamma44 b := by
                rw [sub_mul, smul_mul_assoc, one_mul]; abel
      have h2 : gamma44 b * gamma44 a * gamma44 v - gamma44 v * (gamma44 b * gamma44 a) =
          (2 * zBilin a v) • gamma44 b - (2 * zBilin b v) • gamma44 a := by
        have hav : gamma44 a * gamma44 v =
            (2 * zBilin a v) • (1 : Module.End ℝ Spinor16) - gamma44 v * gamma44 a :=
          eq_sub_of_add_eq hCl
        have hbv : gamma44 b * gamma44 v =
            (2 * zBilin b v) • (1 : Module.End ℝ Spinor16) - gamma44 v * gamma44 b :=
          eq_sub_of_add_eq hCl_bv
        calc gamma44 b * gamma44 a * gamma44 v - gamma44 v * (gamma44 b * gamma44 a)
            = gamma44 b * (gamma44 a * gamma44 v) - gamma44 v * gamma44 b * gamma44 a := by
                simp only [mul_assoc]
          _ = gamma44 b * ((2 * zBilin a v) • 1 - gamma44 v * gamma44 a) -
                gamma44 v * gamma44 b * gamma44 a := by rw [hav]
          _ = (2 * zBilin a v) • gamma44 b - (gamma44 b * gamma44 v) * gamma44 a -
                gamma44 v * gamma44 b * gamma44 a := by
                rw [mul_sub, mul_smul_comm, mul_one]; simp only [mul_assoc]
          _ = (2 * zBilin a v) • gamma44 b -
                ((2 * zBilin b v) • 1 - gamma44 v * gamma44 b) * gamma44 a -
                gamma44 v * gamma44 b * gamma44 a := by rw [hbv]
          _ = (2 * zBilin a v) • gamma44 b - (2 * zBilin b v) • gamma44 a := by
                rw [sub_mul, smul_mul_assoc, one_mul]; abel
      rw [smul_mul_assoc, mul_smul_comm, ← smul_sub]
      have hDiff : (gamma44 a * gamma44 b - gamma44 b * gamma44 a) * gamma44 v -
          gamma44 v * (gamma44 a * gamma44 b - gamma44 b * gamma44 a) =
          (gamma44 a * gamma44 b * gamma44 v - gamma44 v * (gamma44 a * gamma44 b)) -
          (gamma44 b * gamma44 a * gamma44 v - gamma44 v * (gamma44 b * gamma44 a)) := by
        rw [sub_mul, mul_sub]; abel
      rw [hDiff, h1, h2]
      module
    have hApp := congrArg (fun f : Module.End ℝ Spinor16 => (f (0, s)).1) hComm
    simp only [LinearMap.sub_apply, LinearMap.smul_apply, Module.End.mul_apply,
      gamma44, LOp, Prod.fst_sub, Prod.fst_smul, map_zero] at hApp
    have hLHS : spinBivPlus a b (zMul v s) - zMul v (spinBivMinus a b s) =
        zMul (vecBiv a b v) s := by
      have hVecMul : zBilin b v • zMul a s - zBilin a v • zMul b s = zMul (vecBiv a b v) s := by
        ext <;> simp [vecBiv, zMul, dot3, HSub.hSub, Sub.sub, HSMul.hSMul, SMul.smul] <;> try ring
        · intro i; fin_cases i <;> simp [cross3] <;> ring
        · intro i; fin_cases i <;> simp [cross3] <;> ring
      rw [← hVecMul, ← hApp]
      simp only [spinBivPlus, spinBivMinus, LinearMap.smul_apply, LinearMap.sub_apply,
        Module.End.mul_apply, LOp]
      have hLin : ∀ (c : ℝ) (u₁ u₂ : ZornOct ℝ),
          zMul v (c • (u₁ - u₂)) = c • (zMul v u₁ - zMul v u₂) := fun c u₁ u₂ => by
        change LOp v (c • (u₁ - u₂)) = c • (LOp v u₁ - LOp v u₂)
        rw [map_smul, map_sub]
      rw [hLin]
    exact eq_add_of_sub_eq hLHS
  have hDP : ∀ x : ZornOct ℝ,
      (spinBivPlus (quatE 0) (quatE 1) - spinBivPlus (splitEllE 0) (splitEllE 1)) x =
      vecBiv (quatE 0) (quatE 1) x - vecBiv (splitEllE 0) (splitEllE 1) x := by
    intro x
    ext <;> simp [spinBivPlus, vecBiv, LOp, quatE, splitEllE, uPlus, uMinus, zMul, zConj,
      zBilin, dot3, HAdd.hAdd, Add.add, HSub.hSub, Sub.sub, HSMul.hSMul, SMul.smul] <;> try ring
    · intro i; fin_cases i <;> simp [cross3] <;> ring
    · intro i; fin_cases i <;> simp [cross3] <;> ring
  have hDM : ∀ x : ZornOct ℝ,
      (spinBivMinus (quatE 0) (quatE 1) - spinBivMinus (splitEllE 0) (splitEllE 1)) x =
      vecBiv (quatE 0) (quatE 1) x - vecBiv (splitEllE 0) (splitEllE 1) x := by
    intro x
    ext <;> simp [spinBivMinus, vecBiv, LOp, quatE, splitEllE, uPlus, uMinus, zMul, zConj,
      zBilin, dot3, HAdd.hAdd, Add.add, HSub.hSub, Sub.sub, HSMul.hSMul, SMul.smul] <;> try ring
    · intro i; fin_cases i <;> simp [cross3] <;> ring
    · intro i; fin_cases i <;> simp [cross3] <;> ring
  have hDeriv :
      (vecBiv (quatE 0) (quatE 1) (zMul v s) - vecBiv (splitEllE 0) (splitEllE 1) (zMul v s)) =
      zMul (vecBiv (quatE 0) (quatE 1) v - vecBiv (splitEllE 0) (splitEllE 1) v) s +
      zMul v (vecBiv (quatE 0) (quatE 1) s - vecBiv (splitEllE 0) (splitEllE 1) s) := by
    have hT1 := (fun h => h) (show spinBivPlus (quatE 0) (quatE 1) (zMul v s) =
        zMul (vecBiv (quatE 0) (quatE 1) v) s + zMul v (spinBivMinus (quatE 0) (quatE 1) s) from by
      simpa using hTri)
    -- Apply hTri directly to (quatE 0, quatE 1) and (splitEllE 0, splitEllE 1) via subtraction
    rw [← hDP (zMul v s), ← hDM s]
    ext <;> simp [spinBivPlus, spinBivMinus, vecBiv, LOp, quatE, splitEllE, uPlus, uMinus,
      zMul, zConj, zBilin, dot3, HAdd.hAdd, Add.add, HSub.hSub, Sub.sub, HSMul.hSMul, SMul.smul] <;>
      try ring
    · intro i; fin_cases i <;> simp [cross3] <;> ring
    · intro i; fin_cases i <;> simp [cross3] <;> ring
  have hActE0 :
      vecBiv (quatE 0) (quatE 1) (quatE 0) - vecBiv (splitEllE 0) (splitEllE 1) (quatE 0) =
      -quatE 1 := by
    ext <;> simp [vecBiv, quatE, splitEllE, uPlus, uMinus, zBilin, dot3,
      HAdd.hAdd, Add.add, HSub.hSub, Sub.sub, HSMul.hSMul, SMul.smul, Neg.neg] <;>
      (intro i; fin_cases i <;> ring)
  exact ⟨hTri, hDP v, hDM v, hDeriv, hActE0⟩
